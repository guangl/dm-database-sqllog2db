  **Test 1 — `test_watch_03_incremental_appends_only_new_rows`**（WATCH-03 核心）：

  ```
  let tmp = TempDir::new().unwrap();
  let log_path = tmp.path().join("test_watch03.log");
  let db_path = tmp.path().join("test_watch03.db");

  // Phase 1: 写 10 条 + 全文触发
  write_test_log_records(&log_path, 0, 10);
  // 全文路径：用户初始 config——append=false, overwrite=true（清表）
  let mut cfg_initial = build_sqlite_config(&log_path, &db_path);
  cfg_initial.exporter.sqlite.as_mut().unwrap().overwrite = true;
  cfg_initial.exporter.sqlite.as_mut().unwrap().append = false;

  let interrupted = Arc::new(AtomicBool::new(false));
  let pb = build_pb();
  let mut state = WatchLoopState::new(HashMap::new(), Some(db_path.to_string_lossy().into_owned()));
  // 先调用 ensure_offset_table —— 通过公开 API 的方式：handle_watch 启动时已做，
  // 这里直接通过 rusqlite 建表绕过（同 D-05 DDL）。或：先调用一次 trigger_full_file，它会触发 SQLite 建表。

  trigger_full_file(&log_path, &cfg_initial, true, false, &interrupted, &mut state, &pb);
  assert_eq!(count_rows(&db_path, "sqllog_records"), 10, "Phase 1: 全文触发后应有 10 行");

  // Phase 2: 追加 5 条 + 增量触发
  write_test_log_records(&log_path, 10, 5);
  // 增量路径 cfg —— append=true, overwrite=false（trigger_incremental 内部会强制，但显式设置便于阅读）
  let cfg_incremental = build_sqlite_config(&log_path, &db_path);
  trigger_incremental(&log_path, &cfg_incremental, true, false, &interrupted, &mut state, &pb);

  let count_after = count_rows(&db_path, "sqllog_records");
  assert_eq!(count_after, 15, "Phase 2: 增量触发后应为 15 行（不重复），实际 {count_after}");
  ```

  **Test 2 — `test_watch_04_offset_persists_across_restart`**（WATCH-04 核心）：

  ```
  let tmp = TempDir::new().unwrap();
  let log_path = tmp.path().join("test_watch04.log");
  let db_path = tmp.path().join("test_watch04.db");
  let db_url = db_path.to_string_lossy().into_owned();

  // Phase 1: 写 10 + 全文触发（含 save_offset 到 _watch_offsets 表）
  write_test_log_records(&log_path, 0, 10);
  let mut cfg_initial = build_sqlite_config(&log_path, &db_path);
  cfg_initial.exporter.sqlite.as_mut().unwrap().overwrite = true;
  cfg_initial.exporter.sqlite.as_mut().unwrap().append = false;

  let interrupted = Arc::new(AtomicBool::new(false));
  let pb = build_pb();
  let mut state1 = WatchLoopState::new(HashMap::new(), Some(db_url.clone()));
  trigger_full_file(&log_path, &cfg_initial, true, false, &interrupted, &mut state1, &pb);
  let offset_after_full = std::fs::metadata(&log_path).unwrap().len();
  assert!(state1.file_offsets().contains_key(&log_path.canonicalize().unwrap()), "Phase 1: file_offsets 应记录路径");

  // 模拟重启：销毁 state1，通过 offsets::load_offsets 重建。
  // load_offsets 是 pub(super)，跨 crate 不可见——通过 handle_watch 启动逻辑等价路径：
  // 直接用 rusqlite 查询 _watch_offsets 表构造 HashMap，模拟 load_offsets 行为。
  drop(state1);
  let restored_offsets: HashMap<std::path::PathBuf, u64> = {
      let conn = Connection::open(&db_url).unwrap();
      let mut stmt = conn.prepare("SELECT path, byte_offset FROM _watch_offsets").unwrap();
      stmt.query_map([], |row| {
          let p: String = row.get(0)?;
          let o: i64 = row.get(1)?;
          Ok((std::path::PathBuf::from(p), o as u64))
      }).unwrap().filter_map(|r| r.ok()).collect()
  };
  assert!(!restored_offsets.is_empty(), "重启后 _watch_offsets 应非空");

  // Phase 2: 用恢复的 offset 重建 state，追加 7 条 + 增量触发
  let mut state2 = WatchLoopState::new(restored_offsets, Some(db_url.clone()));
  write_test_log_records(&log_path, 10, 7);
  let cfg_incremental = build_sqlite_config(&log_path, &db_path);
  trigger_incremental(&log_path, &cfg_incremental, true, false, &interrupted, &mut state2, &pb);

  let count_after = count_rows(&db_path, "sqllog_records");
  assert_eq!(count_after, 17, "Phase 2: 重启后增量触发应为 17 行（不重复），实际 {count_after}");
  let new_size = std::fs::metadata(&log_path).unwrap().len();
  assert!(new_size > offset_after_full, "文件应已增长");
  ```
