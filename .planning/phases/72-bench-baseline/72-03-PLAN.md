```rust
#[derive(Debug, Deserialize, Clone)]
pub struct SqllogConfig {
    #[serde(default)]
    pub inputs: Vec<String>,           // ← 当前合法字段（v1.12+）
    #[doc(hidden)]
    #[serde(rename = "path", default)]
    pub path_deprecated: Option<toml::Value>,  // ← deprecated 哨兵，validate() 主动报错
}

// validate() 关键约束：
//   - path_deprecated.is_some() → Err（迁移提示）

//   - inputs 含空白条目 → Err
// validate 子命令仅校验 config 结构，不要求 sqllogs/ 目录实际存在于磁盘。
```
