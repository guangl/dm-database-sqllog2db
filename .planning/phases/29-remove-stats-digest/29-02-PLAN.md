    ```rust
    const NEEDS_SPECIAL_NORM: [bool; 256] = {
        let mut t = [false; 256];
        t[b'\'' as usize] = true;
        t[b' ' as usize] = true;
        t[b'\t' as usize] = true;
        t[b'\n' as usize] = true;
        t[b'\r' as usize] = true;
        t[0x0B_usize] = true;
        t[0x0C_usize] = true;
        let mut d = b'0';
        while d <= b'9' { t[d as usize] = true; d += 1; }
        t[b'-' as usize] = true;
        t[b'/' as usize] = true;
        t
    };
    ```
