  目标：验证 `--verbose run` stderr 在摘要之前包含每文件的 `Processed: ... — N records` 行。
  实现：
  ```
  let dir = tempfile::TempDir::new().unwrap();
  let log_dir = dir.path().join("logs");
  std::fs::create_dir_all(&log_dir).unwrap();
  write_test_log(&log_dir.join("a.log"), 5);
  write_test_log(&log_dir.join("b.log"), 7);
  let csv_file = dir.path().join("out.csv");
  // 写一个最小 TOML，引用 log_dir 与 csv_file，logging.file / error.file 指向 temp
  let toml_path = dir.path().join("config.toml");
  std::fs::write(&toml_path, format!(
      "[sqllog]\npath = \"{}\"\n[error]\nfile = \"{}\"\n[logging]\nfile = \"{}\"\nlevel = \"warn\"\nretention_days = 1\n[exporter.csv]\nfile = \"{}\"\noverwrite = true\nappend = false\n",

      dir.path().join("errors.log").to_string_lossy().replace('\\', "/"),
      dir.path().join("app.log").to_string_lossy().replace('\\', "/"),
      csv_file.to_string_lossy().replace('\\', "/"),
  )).unwrap();

  let output = std::process::Command::new(env!("CARGO_BIN_EXE_sqllog2db"))
      .args(["--verbose", "run", "-c", toml_path.to_str().unwrap()])
      .output()
      .expect("failed to spawn");
  assert!(output.status.success(), "verbose run failed: {}", String::from_utf8_lossy(&output.stderr));
  let stderr = String::from_utf8_lossy(&output.stderr);
  let processed_count = stderr.matches("Processed: ").count();
  assert!(
      processed_count &gt;= 2,
      "expected &gt;=2 'Processed: ' lines (one per file), got {processed_count}: {stderr}"
  );
  assert!(
      stderr.contains("5 records") || stderr.contains("7 records"),
      "expected per-file record count in stderr: {stderr}"
  );
  assert!(
      stderr.contains("✓ SQL Log Export Task Completed"),
      "expected completion summary in stderr: {stderr}"
  );
  ```
